\section{栈 (Stack)}

\subsection{代数定义}
栈 $S$ 是定义在元素集 $E$ 上的代数结构，由以下算子与公理刻画：
\begin{itemize}
    \item \textbf{构造算子}：
    \begin{itemize}
        \item $\text{empty}: \to S$ （空栈）
        \item $\text{push}: S \times E \to S$ （压入）
    \end{itemize}
    \item \textbf{观测算子}：
    \begin{itemize}
        \item $\text{pop}: S \to S \cup \{\bot\}$ （弹出）
        \item $\text{top}: S \to E \cup \{\bot\}$ （栈顶元素）
        \item $\text{isEmpty}: S \to \mathbb{B}$ （判空）
    \end{itemize}
\end{itemize}

\textbf{公理系统}（$\forall s \in S, e \in E$）：
\begin{enumerate}
    \item $\text{isEmpty}(\text{empty}) = \text{true}$
    \item $\text{isEmpty}(\text{push}(s, e)) = \text{false}$
    \item $\text{pop}(\text{push}(s, e)) = s$
    \item $\text{top}(\text{push}(s, e)) = e$
    \item $\text{pop}(\text{empty}) = \bot, \quad \text{top}(\text{empty}) = \bot$
\end{enumerate}

\subsection{物理实现方案}
\begin{enumerate}
    \item \textbf{顺序栈}：映射到连续数组 $A[0..m-1]$，栈顶指针 $t \in \{-1, \dots, m-1\}$。$\text{push}$ 操作需满足 $t < m-1$，否则栈溢出。
    \item \textbf{链栈}：映射到单向链表，表头即为栈顶，无容量上限。
\end{enumerate}

\begin{algorithm}[H]
\caption{栈操作伪代码 (Stack Operations)}
\begin{algorithmic}[1]
\State \textbf{数据结构}: 数组 $A[0..m-1]$, 整数 $t \gets -1$
\Procedure{Push}{$e$}
    \If{$t = m - 1$}
        \State \Return \textbf{Error} (Stack Overflow)
    \EndIf
    \State $t \gets t + 1$
    \State $A[t] \gets e$
\EndProcedure
\Procedure{Pop}{}
    \If{$t = -1$}
        \State \Return \textbf{Error} (Stack Underflow)
    \EndIf
    \State $e \gets A[t]$
    \State $t \gets t - 1$
    \State \Return $e$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\section{队列 (Queue)}

\subsection{代数定义}
队列 $Q$ 是定义在元素集 $E$ 上的代数结构：
\begin{itemize}
    \item \textbf{构造算子}：$\text{emptyQ}: \to Q$，$\text{enqueue}: Q \times E \to Q$
    \item \textbf{观测算子}：$\text{dequeue}: Q \to Q \cup \{\bot\}$，$\text{front}: Q \to E \cup \{\bot\}$，$\text{isEmptyQ}: Q \to \mathbb{B}$
\end{itemize}

\textbf{公理系统}（$\forall q \in Q, e \in E$）：
\begin{enumerate}
    \item $\text{isEmptyQ}(\text{emptyQ}) = \text{true}$
    \item $\text{dequeue}(\text{enqueue}(\text{emptyQ}, e)) = \text{emptyQ}$
    \item $\text{dequeue}(\text{enqueue}(\text{enqueue}(q, e_1), e_2)) = \text{enqueue}(\text{dequeue}(\text{enqueue}(q, e_1)), e_2)$
    \item $\text{front}(\text{enqueue}(\text{emptyQ}, e)) = e$
    \item 若 $\text{isEmptyQ}(q) = \text{false}$，则 $\text{front}(\text{enqueue}(q, e)) = \text{front}(q)$
\end{enumerate}

\subsection{循环队列 (Circular Queue)}
为消除假溢出，将数组索引映射到模 $m$ 剩余类环 $\mathbb{Z}_m$。
\begin{itemize}
    \item \textbf{指针移动}：$f \leftarrow (f + 1) \bmod m$，$r \leftarrow (r + 1) \bmod m$。
    \item \textbf{队满判据}：$(r + 1) \bmod m = f$（牺牲一槽位）。
    \item \textbf{队空判据}：$r = f$。
\end{itemize}

\begin{algorithm}[H]
\caption{循环队列操作伪代码 (Circular Queue Operations)}
\begin{algorithmic}[1]
\State \textbf{数据结构}: 数组 $Q[0..m-1]$, 整数 $f \gets 0, r \gets 0$
\Procedure{Enqueue}{$e$}
    \If{$(r + 1) \bmod m = f$}
        \State \Return \textbf{Error} (Queue Full)
    \EndIf
    \State $Q[r] \gets e$
    \State $r \gets (r + 1) \bmod m$
\EndProcedure
\Procedure{Dequeue}{}
    \If{$r = f$}
        \State \Return \textbf{Error} (Queue Empty)
    \EndIf
    \State $e \gets Q[f]$
    \State $f \gets (f + 1) \bmod m$
    \State \Return $e$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\section{字符串 (String)}

\subsection{形式化定义}
\begin{itemize}
    \item \textbf{字母表}：$\Sigma$ 为有限非空字符集合。
    \item \textbf{字符串集合}：$\Sigma^* = \bigcup_{k=0}^{\infty} \Sigma^k$，其中 $\Sigma^k$ 为长度为 $k$ 的字符串集合。
    \item \textbf{空串}：$\varepsilon \in \Sigma^0$。
    \item \textbf{连接运算}：$\circ: \Sigma^* \times \Sigma^* \to \Sigma^*$，满足结合律，且 $\varepsilon$ 为单位元。
    \item \textbf{物理存储}：在底层语言中，字符串 $s \in \Sigma^*$ 映射为字符数组，以终结符 $\# \notin \Sigma$（如 C 语言的 \texttt{\textbackslash 0}）界定边界。
\end{itemize}

\section{高级映射与哈希表体系}

\subsection{字典 (Dictionary) / 映射 (Map)}
\begin{itemize}
    \item \textbf{逻辑定义}：映射关系 $\mathcal{M}: K \rightharpoonup V$，其中 $K$ 为键集，$V$ 为值集，$\rightharpoonup$ 表示偏函数。
    \item \textbf{约束}：$\forall k_1, k_2 \in \text{dom}(\mathcal{M}), \mathcal{M}(k_1) = \mathcal{M}(k_2) \implies k_1 = k_2$（单射性）。
\end{itemize}

\subsection{哈希表 (Hash Table)}
通过哈希函数 $h: K \to \mathbb{Z}_m$ 将键空间压缩至有限数组槽位 $\{0, \dots, m-1\}$。

\subsubsection{(1) 哈希函数}
\begin{itemize}
    \item \textbf{除留余数法}：$h(k) = k \bmod m$，通常取 $m$ 为质数以最小化冲突。
    \item \textbf{理想哈希}：$\forall k_i \neq k_j \implies h(k_i) \neq h(k_j)$。
\end{itemize}

\subsection{哈希碰撞 (Hash Collision)}
\subsubsection{(1) 碰撞必然性}
由鸽巢原理，若 $|K| > m$，则 $\exists k_i \neq k_j$ 使得 $h(k_i) = h(k_j)$。

\subsubsection{(2) 冲突解决方案}
\begin{enumerate}
    \item \textbf{开放寻址法 (Open Addressing)}
    \begin{itemize}
        \item 探测序列：$h_i(k) = (h(k) + \delta(i)) \bmod m$，其中 $\delta(i)$ 为步长函数；所有槽位都存放在主表数组中，冲突时按序列继续探测。

        \item \textbf{线性探测 (Linear Probing)}：$\delta(i) = i$，探测 $h(k), h(k)+1, h(k)+2, \dots$
        \begin{itemize}
            \item 实现最简单、缓存友好，期望探测次数 $O(1 + \alpha)$。
            \item 缺点：易产生\textbf{主聚集 (Primary Clustering)}——冲突键连续堆叠成块，探测链越拖越长，性能急剧恶化。
        \end{itemize}

        \item \textbf{二次探测 (Quadratic Probing)}：$\delta(i) = i^2$，探测 $h(k), h(k)+1, h(k)+4, h(k)+9, \dots$
        \begin{itemize}
            \item 探测间距随 $i$ 增大而拉大，同簇键迅速分散，缓解主聚集（\textbf{次级聚集}仍可能存在）。
            \item 注意：$i^2 \bmod m$ 可能提前循环，仅当 $m$ 为质数且负载因子 $\alpha < 0.5$ 时才能保证遍历全部槽位。
        \end{itemize}

        \item \textbf{双重散列 (Double Hashing)}：$h_i(k) = (h_1(k) + i \cdot h_2(k)) \bmod m$，其中 $h_2(k) \not\equiv 0 \pmod m$
        \begin{itemize}
            \item 典型取法：$h_1(k) = k \bmod m$，$h_2(k) = 1 + (k \bmod (m-1))$，步长随键变化且与 $m$ 互质（$m$ 与 $m-1$ 恒互质）。
            \item 每个键拥有\emph{独立}的探测序列，分布最均匀，最接近理想随机探测，冲突性能最优。
        \end{itemize}

        \item \textbf{开放寻址公共缺陷}：删除不能直接置空（会截断后续键的探测链），须以「墓碑」标记被删槽位；负载因子过高时须\textbf{再散列 (Rehashing)}扩容重建。
    \end{itemize}

    \item \textbf{链地址法 (Chaining)}
    \begin{itemize}
        \item 每个槽位 $i \in \mathbb{Z}_m$ 维护一个链表 $L_i = \{ (k, v) \mid h(k) = i \}$。
        \item 检索时间复杂度：期望 $O(1 + \alpha)$，其中 $\alpha = n/m$ 为负载因子。
        \item \textbf{动态重构}：当 $|L_i| > \tau$（阈值）时，将链表重构为自平衡二叉搜索树（如红黑树），最坏检索复杂度降为 $O(\log n)$。
    \end{itemize}
\end{enumerate}

\begin{algorithm}[H]
\caption{哈希表查找与插入伪代码 (Hash Table Operations)}
\begin{algorithmic}[1]
\State \textbf{数据结构}: 数组 $T[0..m-1]$ (初始为 $\emptyset$), 哈希函数 $h(k)$
\Procedure{Search}{$k$}
    \State $i \gets 0$
    \Repeat
        \State $j \gets (h(k) + i) \bmod m$ \Comment{线性探测}
        \If{$T[j] = \emptyset$ \textbf{or} $T[j].key = k$}
            \State \Return $T[j]$
        \EndIf
        \State $i \gets i + 1$
    \Until{$i = m$}
    \State \Return \textbf{Error} (Not Found)
\EndProcedure
\Procedure{Insert}{$k, v$}
    \State $item \gets \text{Search}(k)$
    \If{$item \neq \textbf{Error}$}
        \State $item.value \gets v$ \Comment{更新}
    \Else
        \State $i \gets 0$
        \While{True}
            \State $j \gets (h(k) + i) \bmod m$
            \If{$T[j] = \emptyset$}
                \State $T[j] \gets \langle k, v \rangle$
                \State \Return
            \EndIf
            \State $i \gets i + 1$
        \EndWhile
    \EndIf
\EndProcedure
\end{algorithmic}
\end{algorithm}
