\chapter{非线性结构 II：图论}

图是一种多对多的网状逻辑结构，用于建模现实世界中复杂的关联网络。

\section{图的形式化定义}

\begin{itemize}
    \item \textbf{对象}：图 $G=(V,E)$，$V$ 有限，$E\subseteq V\times V$。
    \begin{itemize}
        \item 无向图：$E$ 为无序对 $\{\{u,v\}\}$；有向图：$E$ 为有序对 $\{(u,v)\}$。
    \end{itemize}
    \item \textbf{扩展属性}：权重 $w:E\to\mathbb{R}$；邻接 $N: V\to\mathcal{P}(V)$，$N(u)=\{v\mid(u,v)\in E\}$。
\end{itemize}

\section{边的拓扑分类}

\begin{itemize}
    \item \textbf{对象}：有向图 $G$。
    \item \textbf{不变式（DAG）}：不存在非空有向环 $\langle v_1,\dots,v_k\rangle$ 使 $(v_i,v_{i+1})\in E$ 且 $v_1=v_k$。
    \item \textbf{对象}：无向图 $G$。
    \item \textbf{不变式（连通）}：$\forall u,v\in V$，存在路径 $P=\langle u=x_0,\dots,x_k=v\rangle$ 使 $(x_i,x_{i+1})\in E$。
\end{itemize}

\section{连通性深度分析}

\begin{itemize}
    \item \textbf{强连通}：有向图 $\forall u,v$ 存在 $u\to v$ 且 $v\to u$ 有向路径。
    \item \textbf{弱连通}：忽略方向后的无向图连通。
    \item \textbf{桥}：$e\in E$ 为桥 $\iff$ $G'=(V,E\setminus\{e\})$ 的连通分量数严格增加。割点为点对偶。
    \item \textbf{定理}：桥可由 DFS 的 $\text{DFN}/\text{LOW}$ 在 $O(|V|+|E|)$ 内判定（见经典算法章 Tarjan）。
\end{itemize}

\section{图的遍历与降维}

\subsection{遍历规约}

\begin{itemize}
    \item \textbf{对象}：Visited $\subseteq V$，输出序列 $\sigma\in V^*$（数学层以序列代替回调 \texttt{Visit}）。
    \item \textbf{不变式（BFS）}：队列 $F$ 为公平选择器，$\sigma$ 按层扩展；\textbf{（DFS）}：以递归（结构归纳）扩展，$\sigma$ 深度优先。
    \item \textbf{定理}：连通图上 BFS/DFS 均恰访每个可达顶点一次，时间 $O(|V|+|E|)$。
\end{itemize}

\begin{algorithm}[H]
\caption{图遍历规约（以序列 $\sigma$ 代 Visit）}
\begin{algorithmic}[1]
\State \textbf{状态}: $\text{Visited}\gets\emptyset$, $\sigma\gets\langle\rangle$, $F\gets\langle s\rangle$
\Procedure{BFS}{$s$}
    \State $\text{Visited}\gets\{s\}$; $F\gets\langle s\rangle$
    \While{$F\neq\langle\rangle$}
        \State $u\gets\text{popFront}(F)$; $\sigma\gets\sigma\circ\langle u\rangle$
        \ForAll{$v\in N(u)\setminus\text{Visited}$} $\text{Visited}\gets\text{Visited}\cup\{v\}$; $F\gets F\circ\langle v\rangle$ \EndFor
    \EndWhile
\EndProcedure
\Procedure{DFS}{$u$}
    \State $\text{Visited}\gets\text{Visited}\cup\{u\}$; $\sigma\gets\sigma\circ\langle u\rangle$
    \ForAll{$v\in N(u)\setminus\text{Visited}$} \State \text{DFS}($v$) \EndFor
\EndProcedure
\end{algorithmic}
\end{algorithm}

\subsection{拓扑排序}

\begin{itemize}
    \item \textbf{对象}：DAG $G$。
    \item \textbf{不变式}：构造 $\sigma=\langle v_1,\dots,v_n\rangle$ 使 $\forall(u,v)\in E, u=v_i, v=v_j \implies i<j$。
    \item \textbf{定理}：$G$ 有拓扑序 $\iff$ $G$ 为 DAG；Kahn 算法（反复取零入度集）与 DFS 逆后序均为构造性证明，时间 $O(|V|+|E|)$。
\end{itemize}

\section{图的物理存储实现}

\begin{enumerate}
    \item \textbf{邻接矩阵}：$A\in\{0,1\}^{n\times n}$，$A_{ij}=1\iff(v_i,v_j)\in E$；带权 $A_{ij}=w(v_i,v_j)$ 或 $\infty$。空间 $O(|V|^2)$，稠密图实例优。
    \item \textbf{邻接表}：$\text{Adj}[1..n]$，$\text{Adj}[i]=\{v\mid(v_i,v)\in E\}$ 以链表存储。空间 $O(|V|+|E|)$，稀疏图实例优。
\end{enumerate}

\begin{itemize}
    \item \textbf{定理}：两存储在图同构意义下等价，仅时空权衡不同。
\end{itemize}
