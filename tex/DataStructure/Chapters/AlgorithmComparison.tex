\section{核心排序算法算子对比}

\begin{table}[H]
\centering
\footnotesize
\renewcommand{\arraystretch}{1.5}
\begin{tabulary}{\textwidth}{@{}CCCCCCC@{}}
\toprule
\textbf{算法} & \textbf{最好} & \textbf{最坏} & \textbf{平均} & \textbf{空间} & \textbf{稳定性} & \textbf{核心特征} \\

\textbf{快速} & $O(n\log n)$ & $O(n^2)$ & $O(n\log n)$ & $O(\log n)$ & 不稳定 & 分治划分，随机基准 \\
\textbf{归并} & $O(n\log n)$ & $O(n\log n)$ & $O(n\log n)$ & $O(n)$ & 稳定 & 严格分治合并 \\

\textbf{计数} & $O(n+k)$ & $O(n+k)$ & $O(n+k)$ & $O(k)$ & 稳定 & 前缀和寻址，突破比较下界 \\
\textbf{桶} & $O(n+k)$ & $O(n^2)$ & $O(n+k)$ & $O(k)$ & 稳定 & 区间分桶，桶内排序 \\
\textbf{基数} & $O(d(n+k))$ & $O(d(n+k))$ & $O(d(n+k))$ & $O(n+k)$ & 稳定 & 按位稳定排序 \\
\bottomrule

\section{高级图论与检索算子对比}

\begin{table}[H]
\centering
\footnotesize
\renewcommand{\arraystretch}{1.5}
\begin{tabulary}{\textwidth}{@{}CCCCCC@{}}
\toprule
\textbf{算法} & \textbf{目标} & \textbf{时间} & \textbf{空间} & \textbf{核心依赖} & \textbf{限制/判据} \\
\midrule
\textbf{二分查找} & 有序检索 & $O(\log n)$ & $O(1)$ & 顺序存储对切 & 需连续且绝对有序 \\
\textbf{并查集} & 集合维护 & $O(\alpha(n))$ 摊还 & $O(n)$ & 路径压缩 & 接近 $O(1)$ \\
\textbf{Dijkstra} & 单源最短路 & $O(|E|\log|V|)$ & $O(|V|)$ & 优先队列 & 不支持负权 \\
\textbf{A*} & 单源启发 & $O(|E|)$ 期望 & $O(|V|)$ & $f=g+h$ & $h$ 可采纳则最优 \\
\textbf{Bellman-Ford} & 单源带权 & $O(|V||E|)$ & $O(|V|)$ & $|V|-1$ 轮松弛 & 负环判据 \\
\textbf{Floyd-Warshall} & 多源最短路 & $O(|V|^3)$ & $O(|V|^2)$ & 动态规划 & $D[i][i]<0\implies$负环 \\
\textbf{Prim} & MST & $O(|E|\log|V|)$ & $O(|V|)$ & 割定理 & 稠密优 \\
\textbf{Kruskal} & MST & $O(|E|\log|E|)$ & $O(|E|)$ & 并查集 & 稀疏优 \\
\textbf{Kahn} & 拓扑序 & $O(|V|+|E|)$ & $O(|V|)$ & 入度集 & $|\sigma|<|V|\implies$有环 \\
\textbf{Tarjan} & SCC & $O(|V|+|E|)$ & $O(|V|)$ & $DFN/LOW$ & $LOW=DFN\implies$根 \\
\textbf{Dinic} & 最大流 & $O(|V|^2|E|)$ & $O(|V|+|E|)$ & 分层+阻塞流 & 残量无 $s\to t$ 即最优 \\
\bottomrule
\end{tabulary}
\caption{高级图论与检索算子对比}
\label{tab:graph_compare}
\end{table}
