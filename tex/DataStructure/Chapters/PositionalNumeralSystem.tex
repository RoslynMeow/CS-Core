\chapter{位置记数系统}

在数学中，\textbf{$n$ 进制}（进位计数制，Positional Numeral System）是指用一组固定的符号来表示任意实数的计数方法。其严谨的数学形式化定义由以下三个核心要素构成：

\section{任意进制定义}

\subsection{基数与合法数码集 (Alphabet of Digits)}

\begin{itemize}
    \item \textbf{对象}：基数 $n \in \mathbb{Z}^+, n > 1$。
    \item \textbf{不变式}：有限数码集合
$$S_n = \{ x \mid x \in \mathbb{N}, 0 \leq x < n \} = \{0, 1, 2, \dots, n-1\}$$
满足 $|S_n| = n$ 且 $\forall d \in S_n, 0 \le d < n$。
\end{itemize}

\subsection{离散数码序列 (Numeral Sequence)}

\begin{itemize}
    \item \textbf{对象}：双向无穷序列 $\{P_i\}_{i=-\infty}^{\infty}$，约束 $\forall i \in \mathbb{Z}, P_i \in S_n$。
    \item \textbf{不变式（唯一性）}：为使 $y \mapsto \{P_i\}$ 唯一，禁止两类尾：最高有效位左侧不含无穷个 $0$，小数末尾不含无穷个 $(n-1)$（排除 $0.999\ldots = 1.000\ldots$ 二义性）。
    \item \textbf{定理}：在上述不变式下，$\{P_i\} \mapsto y$ 为双射（除零测集外）。
\end{itemize}

\subsection{按权展开映射 (Polynomial Expansion)}

\begin{itemize}
    \item \textbf{对象}：解释映射 $\mathcal{I}_n: S_n^{\mathbb{Z}} \to \mathbb{R}$，
$$y = \mathcal{I}_n(\{P_i\}) = \sum_{i=-\infty}^{\infty} P_i \cdot n^i$$
展开为 $y = \dots + P_2 n^2 + P_1 n^1 + P_0 n^0 + P_{-1} n^{-1} + P_{-2} n^{-2} + \dots$。
    \item \textbf{不变式}：整数部分 $i \ge 0$ 与小数部分 $i < 0$ 同一公式分区，仅指数符号不同。
\end{itemize}

\section{\texorpdfstring{任意进制转换定义 ($n \to m$)}{任意进制转换定义 (n 转 m)}}

\subsection{转换核心公理：值守恒 (Value Conservation)}

\begin{itemize}
    \item \textbf{对象}：源序列 $\{P_i\} \in S_n^{\mathbb{Z}}$，目标序列 $\{Q_j\} \in S_m^{\mathbb{Z}}$。
    \item \textbf{不变式（值守恒）}：
$$y = \sum_{i} P_i n^i = \sum_{j} Q_j m^j, \quad P_i \in S_n, Q_j \in S_m$$
转换是数码表示的变换，实轴值 $y$ 不变。
    \item \textbf{定理}：在唯一性不变式下，满足值守恒的 $\{Q_j\}$ 存在唯一。
\end{itemize}

\subsection{数学求解算子：数码分离 (Digit Extraction Operator)}

\begin{itemize}
    \item \textbf{对象}：已知 $y$，求 $Q_j$ 的闭式算子（全局算子）：
    \begin{itemize}
        \item 整数位 $j \ge 0$：$Q_j = \left\lfloor y / m^j \right\rfloor \bmod m$。
        \item 小数位 $j < 0$：$Q_j = \left\lfloor y_{\text{fraction}} \cdot m^{-j} \right\rfloor \bmod m$，其中 $y_{\text{fraction}} = y - \lfloor y \rfloor \in [0,1)$。
    \end{itemize}
    \item \textbf{不变式}：$\forall j, Q_j \in S_m$。
\end{itemize}

\section{\texorpdfstring{数学算子应用示例 ($10 \to 2$ 进制)}{数学算子应用示例 (10 转 2 进制)}}

\textbf{【示例】} 将 $y = 11.625$ 转换为 $m=2$，验证值守恒。

\subsection{\texorpdfstring{整数部分求解 ($y = 11$, 求解 $j \geq 0$ 的数码)}{(1) 整数部分求解 (y = 11, 求解 j >= 0 的数码)}}

\begin{itemize}
    \item $j=0$: $Q_0 = \lfloor 11/2^0 \rfloor \bmod 2 = 1$
    \item $j=1$: $Q_1 = \lfloor 11/2^1 \rfloor \bmod 2 = \lfloor 5.5 \rfloor \bmod 2 = 1$
    \item $j=2$: $Q_2 = \lfloor 11/2^2 \rfloor \bmod 2 = 0$
    \item $j=3$: $Q_3 = \lfloor 11/2^3 \rfloor \bmod 2 = 1$
    \item $j=4$: $\lfloor 11/2^4 \rfloor = 0$ 终止。得 $(11)_{10} = (1011)_2$，即 $Q_3Q_2Q_1Q_0$。
\end{itemize}

\subsection{\texorpdfstring{小数部分求解 ($y_f = 0.625$, 求解 $j < 0$ 的数码)}{(2) 小数部分求解 (yf = 0.625, 求解 j < 0 的数码)}}

\begin{itemize}
    \item $j=-1$: $Q_{-1} = \lfloor 0.625 \cdot 2^{1} \rfloor \bmod 2 = 1$
    \item $j=-2$: $Q_{-2} = \lfloor 0.625 \cdot 2^{2} \rfloor \bmod 2 = 0$
    \item $j=-3$: $Q_{-3} = \lfloor 0.625 \cdot 2^{3} \rfloor \bmod 2 = 1$
    \item $j=-4$ 起全 $0$ 终止。得 $(0.625)_{10} = (0.101)_2$。
\end{itemize}

\subsection{最终值守恒合并}

$$\sum_{j=-3}^{3} Q_j 2^j = 1\cdot2^3 + 0\cdot2^2 + 1\cdot2^1 + 1\cdot2^0 + 1\cdot2^{-1} + 0\cdot2^{-2} + 1\cdot2^{-3} = 11.625$$
故 $(11.625)_{10} = (1011.101)_2$。

\section{后继与进位 (Successor and Carry)}

\begin{itemize}
    \item \textbf{对象}：序列 $\{P_i\} \in S_n^{\mathbb{Z}}$，进位 $c_i \in \{0,1\}$。
    \item \textbf{不变式}：后继 $\text{succ}: S_n^{\mathbb{Z}} \to S_n^{\mathbb{Z}}$ 定义为 $y \mapsto y+1$ 在数码层的实现。逐位方程：
$$s_i = P_i + c_i, \quad Q_i = s_i \bmod n, \quad c_{i+1} = \left\lfloor s_i / n \right\rfloor, \quad c_0 = 1$$
特例：若 $P_i = n-1$ 则 $Q_i = 0, c_{i+1}=1$ 继续；否则 $Q_i = P_i+1, c_{i+1}=0$ 终止。
    \item \textbf{定理}：$\mathcal{I}_n(\text{succ}(\{P_i\})) = \mathcal{I}_n(\{P_i\}) + 1$，且进位链长度即首个非 $n-1$ 的位置，最坏 $O(k)$。
\end{itemize}

\begin{algorithm}[H]
\caption{后继（进位链）}
\begin{algorithmic}[1]
\State $i \gets 0; c \gets 1$
\While{$c = 1$}
    \State $s \gets P_i + c$; $Q_i \gets s \bmod n$; $c \gets \lfloor s/n \rfloor$; $i \gets i+1$
\EndWhile
\end{algorithmic}
\end{algorithm}

\section{列式加法 (Column Addition)}

\begin{itemize}
    \item \textbf{对象}：$\{P_i\}, \{Q_i\} \in S_n^{\mathbb{Z}}$，和 $\{R_i\} \in S_n^{\mathbb{Z}}$，进位 $c_i \in \{0,1\}$。
    \item \textbf{不变式}：逐列
$$s_i = P_i + Q_i + c_i, \quad R_i = s_i \bmod n, \quad c_{i+1} = \lfloor s_i / n \rfloor, \quad c_0 = 0$$
且 $\forall i, s_i \le 2(n-1)+1 = 2n-1 \implies c_{i+1} \in \{0,1\}$。
    \item \textbf{定理（正确性）}：$\sum_i R_i n^i = \sum_i P_i n^i + \sum_i Q_i n^i$；若末进位 $c_k=1$ 则 $R_k=1$，否则无溢出。
\end{itemize}

\begin{algorithm}[H]
\caption{列式加法}
\begin{algorithmic}[1]
\State $c \gets 0; i \gets 0$
\While{$i < k$}
    \State $s \gets P_i + Q_i + c$; $R_i \gets s \bmod n$; $c \gets \lfloor s/n \rfloor$; $i \gets i+1$
\EndWhile
\If{$c=1$} $R_k \gets 1$ \EndIf
\end{algorithmic}
\end{algorithm}

\section{闭式到递推的细化（进制转换的迭代实现）}

直接计算闭式中的 $m^j$ 需 $O(\log m^j)$ 位与高精度浮点，易引入舍入误差。数学上将其细化为单步递推，仅用 $O(1)$ 的取模/取整。

\subsection{整数项递推：商与余数 (Division-Remainder Iteration)}

\begin{itemize}
    \item \textbf{对象}：整数状态 $y_j \in \mathbb{N}$。
    \item \textbf{状态转移}：
$$\begin{cases}
Q_j = y_j \bmod m \\
y_{j+1} = \left\lfloor y_j / m \right\rfloor
\end{cases}, \quad y_0 = \lfloor y \rfloor$$
    \item \textbf{不变式}：$y = y_{j+1} \cdot m^{j+1} + \sum_{t=0}^{j} Q_t m^t$。
    \item \textbf{终止}：$y_{j+1}=0$。
\end{itemize}

\subsubsection{核心原理解析}
\begin{itemize}
    \item $Q_j = y_j \bmod m$ 隔离最低位；$y_{j+1} = \lfloor y_j/m \rfloor$ 右移一位。
    \item 产出顺序为 $Q_0 \to Q_1 \dots$（低位到高位），序列表示为 $\text{rev}(Q_0\dots Q_{k-1})$。
\end{itemize}

\subsection{小数项递推：积与整数 (Multiplication-Integer Iteration)}

\begin{itemize}
    \item \textbf{对象}：小数状态 $y_j \in [0,1)$。
    \item \textbf{状态转移}：
$$\begin{cases}
Q_j = \lfloor y_j \cdot m \rfloor \\
y_{j-1} = y_j \cdot m - Q_j
\end{cases}, \quad y_{-1} = y - \lfloor y \rfloor$$
    \item \textbf{不变式}：$y_{-1} = \sum_{t=j}^{-1} Q_t m^t + y_{j-1} m^{j}$。
    \item \textbf{终止}：$y_{j-1}=0$ 或达到精度界。
\end{itemize}

\subsubsection{核心原理解析}
\begin{itemize}
    \item $Q_j = \lfloor y_j m \rfloor$ 将小数首位放大到个位摘取；$y_{j-1}=y_j m - Q_j$ 余下纯小数继续。
    \item 产出顺序为 $Q_{-1} \to Q_{-2} \dots$（高位到低位），正序拼接。
\end{itemize}
